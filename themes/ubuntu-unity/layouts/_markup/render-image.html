{{- /* Markdown images. A relative destination resolves against the page bundle and then the site's assets, so `![](photo.jpg)` beside an index.md works. Raster images Hugo can read get width and height, which keeps the reader from jumping as they load. Every image loads lazily and decodes off the critical path. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $dims := dict -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery }}{{ $src = printf "%s?%s" $src . }}{{ end -}} {{- with $u.Fragment }}{{ $src = printf "%s#%s" $src . }}{{ end -}} {{- if reflect.IsImageResourceProcessable . -}} {{- $dims = dict "width" .Width "height" .Height -}} {{- end -}} {{- end -}} {{- end -}} {{- /**/ -}}

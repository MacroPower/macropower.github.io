{{- /* Resolves a URL from front matter the way the Markdown render hooks do. An absolute URL passes through. Anything else resolves to a resource in the page's bundle, then to one under assets/, and otherwise to a path under the site's baseURL, so "/files/cv.pdf" keeps working when the site is served from a subpath. Args (dict): page — the page whose bundle to search url — the URL to resolve */ -}} {{- $u := urls.Parse .url -}} {{- $out := .url -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.page.Resources.Get $path) (resources.Get (strings.TrimPrefix "/" $path)) -}} {{- $out = .RelPermalink -}} {{- else -}} {{- $out = relURL (strings.TrimPrefix "/" $path) -}} {{- end -}} {{- with $u.RawQuery }}{{ $out = printf "%s?%s" $out . }}{{ end -}} {{- with $u.Fragment }}{{ $out = printf "%s#%s" $out . }}{{ end -}} {{- end -}} {{- return $out -}}

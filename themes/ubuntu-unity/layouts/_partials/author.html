{{- /* The site owner as a dict of name, email, born, jobTitle, and focus, read from params.author. A site that sets params.author to a plain string gets that string as the name. The name falls back to the site title, born is normalized to YYYY-MM-DD, and every other missing key is "". Every template that names the owner reads it through here, so call it through partialCached. */ -}} {{- $a := dict "name" "" "email" "" "born" "" "jobTitle" "" "focus" "" -}} {{- with site.Params.author -}} {{- if reflect.IsMap . -}} {{- $m := . -}} {{- range $k, $_ := $a -}} {{- with index $m $k }}{{ $a = merge $a (dict $k (printf "%v" .)) }}{{ end -}} {{- end -}} {{- else -}} {{- $a = merge $a (dict "name" (printf "%v" .)) -}} {{- end -}} {{- end -}} {{- with $a.born }}{{ $a = merge $a (dict "born" ((time.AsTime .).Format "2006-01-02")) }}{{ end -}} {{- if not $a.name }}{{ $a = merge $a (dict "name" site.Title) }}{{ end -}} {{- return $a -}}

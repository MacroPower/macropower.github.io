{{- /* The page's share images as absolute URLs, for Open Graph, Twitter cards, and JSON-LD. The page's images front matter comes first, each entry resolving to a page bundle resource, a file under assets/, or a path under the site's baseURL. Without it, the first bundle image whose name contains feature, cover, or thumbnail stands in, and otherwise the first of params.images. Context: the page. */ -}} {{- $out := slice -}} {{- $names := .Params.images -}} {{- if and $names (not (reflect.IsSlice $names)) }}{{ $names = slice $names }}{{ end -}} {{- if not $names -}} {{- $bundle := .Resources.ByType "image" -}} {{- with or ($bundle.GetMatch "*feature*") ($bundle.GetMatch "{*cover*,*thumbnail*}") -}} {{- $out = slice .Permalink -}} {{- else -}} {{- $names = site.Params.images | default slice -}} {{- if not (reflect.IsSlice $names) }}{{ $names = slice $names }}{{ end -}} {{- $names = first 1 $names -}} {{- end -}} {{- end -}} {{- range $names -}} {{- $out = $out | append (partial "ui/resolve-url.html" (dict "page" $ "url" . "absolute" true)) -}} {{- end -}} {{- return $out -}}

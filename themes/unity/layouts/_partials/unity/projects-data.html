{{- /* The projects the projects page and the Dash show, as a slice of dicts with name, description, language, stars, forks, updated, topics, url, and license. They come from the site's data/projects file, where every key but name is optional: description, language, and updated default to "", and topics to an empty list. An entry without a name warns and drops out. updated becomes YYYY-MM-DD, and a value that is not a date warns and becomes "", so no other template parses a date. Without that file, a site that sets params.projects.github gets that account's public repositories from the GitHub API at build time: sources with a language, no forks or archived repositories, most starred first. Set the HUGO_GITHUB_TOKEN environment variable to lift the API's rate limit for anonymous requests. A failed request warns and yields no projects rather than failing the build. Hugo's file cache keeps a fetched resource forever by default, so the request's cache key holds the build's date. The first build of each day fetches the list again, and later builds that day read the cache. An offline build on a new day warns and yields no projects. The request fails after 10 seconds, so a network that drops packets cannot stall the build. Curation params then narrow the list from either source. params.projects.exclude names the projects to leave out, and params.projects.languages names the languages to keep. Each takes a list or one string and ignores case. Setting params.projects.requireDescription to true leaves out every project without a description. The result is the same on every page, so call it through partialCached. */ -}} {{- $projects := slice -}} {{- with hugo.Data.projects -}} {{- range . -}} {{- if not .name -}} {{- warnf "unity: data/projects has an entry without a name: %v" . -}} {{- continue -}} {{- end -}} {{- $p := merge . (dict "description" (.description | default "") "language" (.language | default "") "topics" (.topics | default slice)) -}} {{- $updated := "" -}} {{- with $p.updated -}} {{- $updated = partial "unity/ui/iso-date.html" . -}} {{- if not $updated -}} {{- warnf "unity: the data/projects entry %q sets updated to %q, which is not a date such as 2024-05-20" (printf "%v" $p.name) (printf "%v" .) -}} {{- end -}} {{- end -}} {{- $projects = $projects | append (merge $p (dict "updated" $updated)) -}} {{- end -}} {{- else -}} {{- with site.Params.projects.github -}} {{- $url := printf "https://api.github.com/users/%s/repos?per_page=100&type=owner&sort=pushed" (urlquery .) -}} {{- $headers := dict "Accept" "application/vnd.github+json" "X-GitHub-Api-Version" "2022-11-28" -}} {{- with os.Getenv "HUGO_GITHUB_TOKEN" }}{{ $headers = merge $headers (dict "Authorization" (printf "Bearer %s" .)) }}{{ end -}} {{- $opts := dict "headers" $headers "key" (print $url (now.Format "2006-01-02")) "timeout" "10s" -}} {{- with try (resources.GetRemote $url $opts) -}} {{- with .Err -}} {{- warnf "unity: cannot fetch the GitHub repositories for params.projects.github: %s" . -}} {{- else with .Value -}} {{- range . | transform.Unmarshal -}} {{- if or .fork .archived (not .language) }}{{ continue }}{{ end -}} {{- $p := dict "name" .name "description" (.description | default "") "language" .language "stars" .stargazers_count "forks" .forks_count "updated" (partial "unity/ui/iso-date.html" .pushed_at) "topics" (.topics | default slice) "url" .html_url -}} {{- with .license }}{{ with .spdx_id }}{{ if ne . "NOASSERTION" }}{{ $p = merge $p (dict "license" .) }}{{ end }}{{ end }}{{ end -}} {{- $projects = $projects | append $p -}} {{- end -}} {{- $projects = sort (sort $projects "name") "stars" "desc" -}} {{- else -}} {{- warnf "unity: GitHub has no account %q for params.projects.github" site.Params.projects.github -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $cfg := site.Params.projects -}} {{- $exclude := $cfg.exclude | default slice -}} {{- if not (reflect.IsSlice $exclude) }}{{ $exclude = slice $exclude }}{{ end -}} {{- $languages := $cfg.languages | default slice -}} {{- if not (reflect.IsSlice $languages) }}{{ $languages = slice $languages }}{{ end -}} {{- if or $exclude $languages $cfg.requireDescription -}} {{- $skip := slice -}} {{- range $exclude }}{{ $skip = $skip | append (lower (printf "%v" .)) }}{{ end -}} {{- $keep := slice -}} {{- range $languages }}{{ $keep = $keep | append (lower (printf "%v" .)) }}{{ end -}} {{- $curated := slice -}} {{- range $projects -}} {{- if in $skip (lower (printf "%v" .name)) }}{{ continue }}{{ end -}} {{- if and $keep (not (in $keep (lower (printf "%v" .language)))) }}{{ continue }}{{ end -}} {{- if and $cfg.requireDescription (not .description) }}{{ continue }}{{ end -}} {{- $curated = $curated | append . -}} {{- end -}} {{- $projects = $curated -}} {{- end -}} {{- return $projects -}}

{{- /* The site's social links from params.social, as a slice of {name, url, short} dicts in configuration order. The documented form is a list of tables, and a table of name = "url" pairs, the form other themes use, also works, sorted by name. Entries without a url drop out. An entry whose url unity/ui/script-scheme.html names, such as a javascript: url, drops out with a warning, because the home terminal's script opens each url. Call it through partialCached. */ -}} {{- $all := slice -}} {{- with site.Params.social -}} {{- if reflect.IsMap . -}} {{- range $name, $url := . -}} {{- $all = $all | append (dict "name" $name "url" (printf "%v" $url) "short" "") -}} {{- end -}} {{- else if reflect.IsSlice . -}} {{- range . -}} {{- if and (reflect.IsMap .) .url -}} {{- $all = $all | append (dict "name" (.name | default "") "url" .url "short" (.short | default "")) -}} {{- end -}} {{- end -}} {{- else -}} {{- warnf `unity: params.social should be a list of {name, url} tables or a table of name = "url" pairs` -}} {{- end -}} {{- end -}} {{- $out := slice -}} {{- range $s := $all -}} {{- with partial "unity/ui/script-scheme.html" $s.url -}} {{- warnf "unity: the params.social entry %q links a %s: url, which the theme leaves out." $s.name . -}} {{- else -}} {{- $out = $out | append $s -}} {{- end -}} {{- end -}} {{- return $out -}}

{{- /* The site's social links from params.social, as a slice of {name, url, short} dicts in configuration order. The documented form is a list of tables, and a table of name = "url" pairs, the form other themes use, also works, sorted by name. Entries without a url drop out. Call it through partialCached. */ -}} {{- $out := slice -}} {{- with site.Params.social -}} {{- if reflect.IsMap . -}} {{- range $name, $url := . -}} {{- $out = $out | append (dict "name" $name "url" (printf "%v" $url) "short" "") -}} {{- end -}} {{- else if reflect.IsSlice . -}} {{- range . -}} {{- if and (reflect.IsMap .) .url -}} {{- $out = $out | append (dict "name" (.name | default "") "url" .url "short" (.short | default "")) -}} {{- end -}} {{- end -}} {{- else -}} {{- warnf "unity: params.social should be a list of {name, url} tables" -}} {{- end -}} {{- end -}} {{- return $out -}}

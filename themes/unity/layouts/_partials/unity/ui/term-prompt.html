{{- /* A desktop-fiction shell prompt: "handle@host:path$[ cmd]". The handle comes from unity/ui/author-handle.html, and the host is params.desktop.host, which defaults to "unity", the machine-name fiction named for the desktop itself. Args (dict): path — the prompt's cwd (e.g. "~/blog"); cmd — optional command rendered after the prompt. */ -}} {{- $host := site.Params.desktop.host | default "unity" -}} {{- $prompt := printf "%s@%s:%s$" (partialCached "unity/ui/author-handle.html" .) $host .path -}} {{- with .cmd }}{{ $prompt = printf "%s %s" $prompt . }}{{ end -}} {{- return $prompt -}}

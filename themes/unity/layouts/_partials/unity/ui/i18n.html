{{- /* The strings the scripts write at runtime, translated for the page's language. Every key in the theme's i18n/en.toml that starts with js_ goes through T, so a site's i18n/.toml translates them like any template string. unity/ui/i18n-en.html reads that en.toml to list the keys. Hugo falls back only to the default language's table for a key that a translation lacks, and the theme's en.toml fills only the en table. So on a site whose default language is not en, a regional English such as en-gb included, a missing key renders empty. This partial gives such a js_ key its English text, as unity/ui/t.html does for template strings. It warns once per language with every key that falls back, under an id that a site can list in ignoreLogs. It skips the warning for a language whose key is en or a regional English such as en-gb, because the theme's English is the right text there. A plural key renders at every count from 0 to 199 under the plural rules of whichever translation file matched, which only Hugo knows. The output keeps the distinct forms and an index string with one digit per count naming its form. assets/unity/js/i18n.ts reads larger counts at 100 plus the count modulo 100, where CLDR's integer rules repeat. Returns a dict of strings and plurals, which unity/ui/runtime.html publishes as window.UP_I18N once per language. */ -}} {{- $strings := dict -}} {{- $plurals := dict -}} {{- $missing := slice -}} {{- $english := eq (index (split (replace (lower site.Language.Name) "_" "-") "-") 0) "en" -}} {{- range $key, $v := partialCached "unity/ui/i18n-en.html" . -}} {{- if not (T $key) }}{{ $missing = $missing | append $key }}{{ end -}} {{- if hasPrefix $key "js_" -}} {{- if and (reflect.IsMap $v) (isset $v "one") -}} {{- $forms := slice -}} {{- $seen := dict -}} {{- $index := "" -}} {{- range seq 0 199 -}} {{- $text := T $key . | default (cond (eq . 1) $v.one $v.other) -}} {{- if not (isset $seen $text) -}} {{- $seen = merge $seen (dict $text (len $forms)) -}} {{- $forms = $forms | append $text -}} {{- end -}} {{- $index = printf "%s%d" $index (index $seen $text) -}} {{- end -}} {{- $plurals = merge $plurals (dict $key (dict "forms" $forms "index" $index)) -}} {{- else -}} {{- $strings = merge $strings (dict $key (T $key | default (cond (reflect.IsMap $v) $v.other $v))) -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $english -}} {{- with $missing -}} {{- $list := delimit (first 8 .) ", " -}} {{- if gt (len .) 8 }}{{ $list = printf "%s, and %d more" $list (sub (len .) 8) }}{{ end -}} {{- warnidf "unity-i18n-missing" "unity: the %q translation lacks %d of the theme's i18n keys, so pages show those strings in English: %s. Copy the missing keys from the theme's i18n/en.toml." site.Language.Name (len .) $list -}} {{- end -}} {{- end -}} {{- return dict "strings" $strings "plurals" $plurals -}}

{{- /* The language indicator, after Unity's keyboard-layout badge. It renders only on a multilingual site. The badge shows the current language's params.desktop.languageBadge, or else the first two letters of its name, and the dropdown lists every language, each row leading to this page's translation or, when the page has none, to that language's home page. The rows are links, which the HUD indexes and the panel's arrow-key roving reaches. Context: the page. */ -}} {{- if hugo.IsMultilingual -}} {{- $page := . -}} {{- $current := site.Language.Name -}} {{- $rows := slice -}} {{- range site.Home.AllTranslations -}} {{- $lang := .Language -}} {{- $target := . -}} {{- range $page.AllTranslations }}{{ if eq .Language.Name $lang.Name }}{{ $target = . }}{{ end }}{{ end -}} {{- $rows = $rows | append (dict "label" (or $lang.Label $lang.Name) "locale" $lang.Locale "url" $target.RelPermalink "current" (eq $lang.Name $current)) -}} {{- end }}

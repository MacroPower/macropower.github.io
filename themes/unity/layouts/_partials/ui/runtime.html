It sets window.UP_SITE, the site facts the scripts read, and window.UP_I18N, the strings from ui/i18n.html. assets/unity/js/types.ts declares both. An external file keeps the theme working under a Content Security Policy without 'unsafe-inline', and browsers cache it across pages. baseof.html renders it before the bundles, which run in document order. "repo" (site.Params.repoUrl | default "") "rss" $rss "hotspot" (site.Params.desktop.hotspot | default "personal-site") "nav" $nav -}} {{- $js := printf "window.UP_SITE=%s;window.UP_I18N=%s;" ($site | jsonify) (partial "ui/i18n.html" . | jsonify) -}} {{- $res := resources.FromString (printf "js/runtime.%s.js" site.Language.Name) $js | minify | fingerprint -}}

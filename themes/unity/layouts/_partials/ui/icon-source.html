{{- /* Reads assets/unity/icons/.svg once per build for icon.html. Returns a dict of svg, the file's markup, and ids, the ids the file both defines and references through url(#id) or href="#id". Call it through partialCached keyed on the name. */ -}} {{- $out := dict -}} {{- with resources.Get (printf "unity/icons/%s.svg" .) -}} {{- $c := .Content -}} {{- $refs := slice -}} {{- range findRE `url\(#[^)]+\)` $c -}} {{- $refs = $refs | append (. | strings.TrimPrefix "url(#" | strings.TrimSuffix ")") -}} {{- end -}} {{- range findRE `href="#[^"]+"` $c -}} {{- $refs = $refs | append (. | strings.TrimPrefix `href="#` | strings.TrimSuffix `"`) -}} {{- end -}} {{- $ids := slice -}} {{- range $refs | uniq -}} {{- if strings.Contains $c (printf `id="%s"` .) }}{{ $ids = $ids | append . }}{{ end -}} {{- end -}} {{- $out = dict "svg" $c "ids" $ids -}} {{- else -}} {{- errorf "icon.html: unity/icons/%s.svg not found" . -}} {{- end -}} {{- return $out -}}
